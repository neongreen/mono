📊 Demo 6: Memory Monitoring and Auto-Recovery

Proactive monitoring and automatic iframe restart on high memory usage.

⚠️ Note: The performance.memory API is non-standard and only available in Chrome/Chromium browsers. This demo won't work in other browsers.

Strategy: Monitor + React

When cross-origin isolation isn't possible, monitor memory usage and automatically restart iframes before they cause problems. This is a reactive approach used by CodeSandbox and other platforms.

// Memory monitoring with auto-restart
class IframeMemoryMonitor {
  constructor(iframe, options = {}) {
    this.iframe = iframe;
    this.threshold = options.threshold || 80; // percent
    this.checkInterval = options.interval || 2000; // ms
    this.maxRestarts = options.maxRestarts || 3;
    this.restartCount = 0;
    this.monitoring = false;
  }
  
  start() {
    this.monitoring = true;
    this.intervalId = setInterval(() => {
      this.checkMemory();
    }, this.checkInterval);
  }
  
  stop() {
    this.monitoring = false;
    clearInterval(this.intervalId);
  }
  
  checkMemory() {
    if (!performance.memory) {
      console.warn('performance.memory not available');
      return;
    }
    
    const { usedJSHeapSize, jsHeapSizeLimit } = performance.memory;
    const usage = (usedJSHeapSize / jsHeapSizeLimit) * 100;
    
    if (usage > this.threshold && this.restartCount < this.maxRestarts) {
      console.warn(\`High memory: \${usage.toFixed(1)}%\`);
      this.restartIframe();
    }
  }
  
  restartIframe() {
    this.restartCount++;
    console.log(\`Restarting iframe (attempt \${this.restartCount})\`);
    
    const src = this.iframe.src;
    this.iframe.src = 'about:blank';
    
    setTimeout(() => {
      this.iframe.src = src;
      // Reset counter after successful recovery
      setTimeout(() => {
        this.restartCount = Math.max(0, this.restartCount - 1);
      }, 60000); // Reset after 1 minute
    }, 100);
  }
}

// Usage
const monitor = new IframeMemoryMonitor(iframe, {
  threshold: 80,
  interval: 2000,
  maxRestarts: 3
});

monitor.start();
